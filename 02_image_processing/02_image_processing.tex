\begin{frame}
    \frametitle{Derivatives of Images}
    \begin{block}{Continuous derivative}
        \begin{equation*}
            f'(x) = \lim_{h\rightarrow0}\frac{f(x+h)-f(x)}{h}
        \end{equation*}
    \end{block}
    \begin{block}{Discrete derivatives}
        Several choices:

        \begin{center}
            \begin{tabularx}{0.8\textwidth}{lX}
                forward difference                    & $\Delta_x f(x) = f(x+1) - f(x)$   \\
                \rule{0pt}{1.5em}central difference   & $\delta_x f(x) = f(x+1) - f(x-1)$ \\
                \rule{0pt}{1.5em}backwards difference & $\nabla_x f(x) = f(x) - f(x-1)$
            \end{tabularx}
        \end{center}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Image Filtering}
    \begin{block}{Filters as operators}
        A filter $\mathcal{H}$ can be applied on an image $f$:
        \begin{center}
            $\mathcal{H}(f(x,y)) = r(x,y)$
        \end{center}
        It can have the same properties as in 1-D:
        \bigskip

        \begin{tabularx}{\textwidth}{lX}
            \multirow{2}{*}{linearity} & \rule{0pt}{1.25em}$\mathcal{H}\lbrace\alpha\cdot f(x,y)\rbrace = \alpha\cdot
                \mathcal{H}\lbrace f(x,y)\rbrace$                                                                              \\
                                       & \rule{0pt}{1.25em}$\mathcal{H}\lbrace f_1(x,y) + f_2(x,y)\rbrace = \mathcal{H}\lbrace
                f_1(x,y)\rbrace + \mathcal{H}\lbrace f_2(x,y)\rbrace$                                                          \\
            shift-invariance           & \rule{0pt}{2em}$\mathcal{H}\lbrace (f(x-x_0, y-y_0)\rbrace = r(x-x_0, y-y_0)$
        \end{tabularx}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Linear, Shift-invariant Filters}
    \begin{block}{Filter kernels}
        Linear, shift-invariant filters are characterized by \textbf{filter kernels}
        $k$ and are applied to an image by a convolution.
    \end{block}
    \begin{block}{Convolution with a filter}
        Image $f$, filter kernel $k$\vspace{-0.5\bigskipamount}
        \begin{equation*}
            \mathcal{H}\lbrace f\rbrace(x,y) = f\ast k = \sum_{i=-\frac{w_k}{2}}^{\frac{w_k}{2}}
            \sum_{j=-\frac{h_k}{2}}^{\frac{h_k}{2}} f(x-i, y-j)\cdot k(i, j)
        \end{equation*}
    \end{block}\vspace{-0.5\bigskipamount}
\end{frame}

\begin{frame}
    \frametitle{Efficient Filtering}
    \begin{block}{Convolution in the Fourier domain}
        For large images and /  or filters:
        \begin{enumerate}
            \item Transform $f$ and $k$ to the Fourier domain
                  \begin{align*}
                      F & =	\mathcal{F}(f) & K & = \mathcal{F}(k)
                  \end{align*}
            \item Like in 1-D, convolution then is a simple multiplication
                  \begin{equation*}
                      \mathcal{F}\lbrace f\ast k\rbrace = F\cdot K
                  \end{equation*}
            \item Get the filtered image by transforming the result back to the
                  spatial domain
        \end{enumerate}
        % $\tilde{w}=\mathcal{F}(w),\, \tilde{g}=\mathcal{F}(g)$
        % \begin{equation*}
        %   \tilde{g}' = \underbrace{\tilde{w}\cdot\tilde{g}}_{\text{simple multiplication}}
        % \end{equation*}
    \end{block}

\end{frame}

\begin{frame}[c]{Segmentation}
    \begin{block}{Goal}
        Separation of regions in images:
        \begin{itemize}
            \item Organs
            \item Bones
        \end{itemize}
    \end{block}
    \begin{block}{Thresholding}
        Image $f$, segmented image $t$, threshold $\theta$

\begin{frame}{Why Deep Learning for Medical Image Analysis}
    Medical images contain high intra-class variability (anatomy, pathology, acquisition site, scanner vendor, field strength) and strong spatial correlations. Classical filters capture only a narrow family of patterns, while convolutional networks learn hierarchical descriptors adapted to the task and the modality.

    Typical tasks addressed with deep learning include tissue and lesion segmentation, disease classification, landmark detection, registration, reconstruction from undersampled acquisitions, and synthesis across modalities. The empirical gains are most pronounced when annotated data, compute, and well-designed evaluation are aligned.
\end{frame}

\begin{frame}{The Artificial Neuron}
    A single unit computes an affine transform followed by a non-linearity:
    \begin{equation*}
        a = \sigma\left(\sum_{i=1}^{d} w_i x_i + b\right) = \sigma(\mathbf{w}^\top \mathbf{x} + b)\,.
    \end{equation*}
    \begin{center}
        \begin{tikzpicture}[>=latex, every node/.style={font=\footnotesize}]
            \foreach \i/\lbl in {1/$x_1$,2/$x_2$,3/$x_3$,4/$x_4$}{
                \node[draw, circle, minimum size=0.6cm] (x\i) at (0,{2-0.8*\i}) {\lbl};
            }
            \node[draw, circle, fill=faublue!20, minimum size=0.9cm] (n) at (3.5,-0.4) {$\Sigma,\sigma$};
            \node[right=0.8cm of n] (out) {$a$};
            \foreach \i/\w in {1/$w_1$,2/$w_2$,3/$w_3$,4/$w_4$}{
                \draw[->] (x\i) -- node[above, sloped] {\w} (n);
            }
            \draw[->] (n) -- (out);
            \node[below=0.2cm of n] {bias $b$};
        \end{tikzpicture}
    \end{center}
    Stacking such units with learnable weights $\mathbf{W}^{(\ell)}$ and biases $\mathbf{b}^{(\ell)}$ produces a multi-layer perceptron (MLP). Depth makes it possible to approximate non-linear functions that no single layer can represent compactly.
\end{frame}

\begin{frame}{Activation Functions}
    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.55\textwidth}
            Non-linearities break the linearity of stacked affine maps. Common choices:
            \begin{itemize}
                \item Sigmoid $\sigma(z) = \tfrac{1}{1+e^{-z}}$: saturates, suffers from vanishing gradients.
                \item Hyperbolic tangent $\tanh(z)$: zero-centred but still saturating.
                \item Rectified linear unit $\mathrm{ReLU}(z) = \max(0, z)$: sparse, fast, dominant in vision.
                \item Leaky ReLU and GELU: address dead units and smooth the gradient at the origin.
                \item Softmax: normalises the output of a classification head into a categorical distribution.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{tikzpicture}
                \begin{axis}[width=\textwidth, height=5cm, xmin=-4, xmax=4, ymin=-1.2, ymax=1.5,

\end{frame}

\begin{frame}{Training Objective and Loss Functions}
    For a dataset $\{(\mathbf{x}_i, \mathbf{y}_i)\}_{i=1}^N$ and a network $f_\theta$ we minimise the empirical risk
    \begin{equation*}
        \mathcal{L}(\theta) = \frac{1}{N} \sum_{i=1}^N \ell\bigl(f_\theta(\mathbf{x}_i), \mathbf{y}_i\bigr) + \lambda\, \Omega(\theta)\,.
    \end{equation*}
    Common losses for imaging tasks are cross-entropy for classification, mean squared error and $\ell_1$ for regression and reconstruction, and Dice or combined Dice plus cross-entropy for segmentation. The regulariser $\Omega$ typically includes weight decay and implicit terms such as dropout, batch normalisation, or augmentation.
\end{frame}

\begin{frame}{Gradient Descent and Backpropagation}
    Parameters are updated by stochastic gradient descent
    \begin{equation*}
        \theta^{(t+1)} = \theta^{(t)} - \eta\, \nabla_\theta \mathcal{L}_{\mathcal{B}}(\theta^{(t)})\,,
    \end{equation*}
    where $\mathcal{B}$ is a mini-batch and $\eta$ is the learning rate. The gradient is computed by backpropagation, the application of the chain rule along the computational graph.
    \begin{center}
        \begin{tikzpicture}[>=latex, every node/.style={font=\footnotesize}]
            \node[faunode, minimum width=1.2cm] (x) {$\mathbf{x}$};
            \node[faunode, right=0.5cm of x, minimum width=1.2cm] (h1) {layer $1$};
            \node[faunode, right=0.5cm of h1, minimum width=1.2cm] (h2) {layer $2$};
            \node[faunode, right=0.5cm of h2, minimum width=1.2cm] (h3) {layer $L$};
            \node[faunode, right=0.5cm of h3, minimum width=1.2cm] (loss) {$\ell$};
            \draw[fauarrow] (x) -- (h1);
            \draw[fauarrow] (h1) -- (h2);
            \draw[fauarrow] (h2) -- node[above] {$\cdots$} (h3);
            \draw[fauarrow] (h3) -- (loss);
            \draw[->, red, thick] (loss.south) to[bend left=25] node[below, font=\tiny, red] {$\partial \ell / \partial \theta$} (x.south);
        \end{tikzpicture}
    \end{center}
    In practice Adam or SGD with momentum, learning-rate schedules, and gradient clipping are used to stabilise optimisation on non-convex landscapes.
\end{frame}

\begin{frame}{Convolution in the Network}
    For input tensor $X \in \mathbb{R}^{C_\mathrm{in}\times H\times W}$ and kernel $K \in \mathbb{R}^{C_\mathrm{out}\times C_\mathrm{in}\times k\times k}$
    \begin{equation*}
        Y_{c,i,j} = \sum_{c'=1}^{C_\mathrm{in}} \sum_{u=-r}^{r} \sum_{v=-r}^{r} K_{c,c',u,v}\, X_{c',\,i+u,\,j+v} + b_c\,.
    \end{equation*}
    Volumetric data use the 3-D analogue with a fourth sum over slices. Parameters per layer equal $C_\mathrm{out}\, C_\mathrm{in}\, k^3$ for 3-D convolutions, which makes kernel design (usually $3{\times}3{\times}3$) a central decision in medical imaging architectures.
\end{frame}

\begin{frame}{Stride, Padding and Receptive Field}
    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.55\textwidth}
            The output spatial size for a kernel of size $k$, stride $s$, and padding $p$ is
            \begin{equation*}
                H_\mathrm{out} = \left\lfloor \frac{H_\mathrm{in} + 2p - k}{s}\right\rfloor + 1\,.
            \end{equation*}
            Padding preserves resolution, stride aggressively reduces it. The receptive field grows approximately linearly with depth for stride one and exponentially when combined with pooling. In segmentation networks a sufficiently large receptive field is essential to see whole organs, while keeping spatial resolution for precise boundaries.
        \end{column}

\begin{frame}{Batch Normalisation and Weight Initialisation}
    Training deep networks is highly sensitive to the scale of activations. Batch normalisation whitens the pre-activation in each mini-batch,
    \begin{equation*}
        \hat{z} = \frac{z - \mu_\mathcal{B}}{\sqrt{\sigma^2_\mathcal{B} + \varepsilon}}\,,\qquad z' = \gamma\hat{z} + \beta\,,
    \end{equation*}
    where $\gamma, \beta$ are learned. Alternatives include layer, group, and instance normalisation; for small batch sizes, which are common in 3-D medical imaging, group normalisation is usually preferred. Initialisation schemes such as He or Xavier set the variance of weights so that the activation variance remains stable across depth.
\end{frame}

\begin{frame}{Regularisation and Generalisation}
    Networks have millions of parameters and can memorise small datasets. Regularisation prevents overfitting:
    \begin{itemize}
        \item Weight decay, the $\ell_2$ penalty on parameters.
        \item Dropout, which zeroes a fraction of activations at training time.
        \item Early stopping based on validation loss.
        \item Data augmentation, the main driver of generalisation in medical imaging.
        \item Stochastic depth and label smoothing.
    \end{itemize}
    In imaging, augmentation includes elastic deformations, random rotations, intensity shifts, bias field simulation for magnetic resonance imaging (MRI), and acquisition-specific augmentations such as k-space subsampling or CT noise modelling.
\end{frame}

\begin{frame}{Residual Connections}

\begin{frame}{Segmentation Loss Functions}
    Class imbalance is the rule in medical imaging because lesions or organs of interest occupy a small fraction of the volume. Pixel-wise cross-entropy is easily dominated by background, so overlap-based losses are preferred.

    The soft Dice loss for a predicted probability map $p$ and a binary mask $g$ is
    \begin{equation*}
        \mathcal{L}_\mathrm{Dice} = 1 - \frac{2\sum_i p_i g_i + \epsilon}{\sum_i p_i + \sum_i g_i + \epsilon}\,.
    \end{equation*}
    A combined loss $\mathcal{L} = \mathcal{L}_\mathrm{Dice} + \mathcal{L}_\mathrm{CE}$ is a robust default. For highly imbalanced problems the Tversky and focal losses reweight false negatives or hard examples.
\end{frame}

\begin{frame}{Transposed Convolution and Upsampling}
    The decoder restores resolution either by nearest neighbour or bilinear interpolation followed by a standard convolution, or by transposed convolution. Transposed convolution is equivalent to inserting zeros between input samples and applying a normal convolution, producing a learned upsampling operator.

    Transposed convolutions can introduce checkerboard artefacts. Resize-then-convolve approaches, used for example in nnU-Net, are often more stable in medical imaging where sharp quantitative boundaries matter.
\end{frame}

\begin{frame}{Attention Mechanisms and Transformers}
    Self-attention computes pairwise affinities between tokens and aggregates features accordingly,
    \begin{equation*}
        \mathrm{Attn}(Q, K, V) = \mathrm{softmax}\!\left(\frac{QK^\top}{\sqrt{d_k}}\right) V\,.
    \end{equation*}
    Vision transformers (ViT) tokenise an image into patches, linearly embed them, and apply stacks of multi-head self-attention blocks. Hybrid networks such as Swin-UNETR combine a windowed transformer encoder with a convolutional decoder and are competitive for 3-D segmentation.

    Attention modules are also widely used as add-ons to convolutional networks, either spatial (attention gates in U-Net) or channel-wise (squeeze-and-excitation).
\end{frame}

\begin{frame}{Training Protocol for Medical Imaging}
    A rigorous protocol avoids optimistic bias. The dataset is split at the patient level, never at the slice or volume level, to prevent leakage. Stratification by site, scanner, and class balance is standard. Cross-validation, typically five-fold, is used during model development, and a held-out test set is reserved for final reporting.
    \begin{center}
        \begin{tikzpicture}[every node/.style={font=\footnotesize}]

            \node at (3, 0.3) {all subjects};
            \draw[fill=faublue!30] (0,-0.9) rectangle (4,-0.3);
            \node at (2,-0.6) {training and validation (cross-validation)};
            \draw[fill=red!30] (4,-0.9) rectangle (6,-0.3);

        \end{tikzpicture}
    \end{center}
    External validation on data from other institutions is the gold standard for any claim of clinical usefulness. Reporting should include confidence intervals and per-centre performance.
\end{frame}

\begin{frame}{Data Augmentation for Medical Imaging}
    Deep networks generalise only along directions covered by the training distribution. Augmentation enlarges that distribution in a controlled way:
    \begin{itemize}
        \item Geometric: random affine, elastic deformation, anisotropic resampling, mirroring when anatomically plausible.
        \item Intensity: gamma, brightness, contrast, Gaussian noise, bias field simulation.
        \item Modality-specific: k-space undersampling and motion for MRI, photon starvation and beam hardening for CT, speckle noise for ultrasound.
        \item Test-time augmentation and ensembling to stabilise predictions.
    \end{itemize}
    Implementations such as TorchIO, MONAI transforms, and nnU-Net provide validated pipelines.
\end{frame}

\begin{frame}{Evaluation Metrics}
    Metrics must reflect the clinical question. For segmentation, the Dice similarity coefficient and the Jaccard index measure overlap,
    \begin{equation*}
        \mathrm{DSC}(A, B) = \frac{2|A \cap B|}{|A| + |B|}\,,\qquad \mathrm{IoU}(A, B) = \frac{|A \cap B|}{|A \cup B|}\,,
    \end{equation*}
    while the Hausdorff distance and the average surface distance quantify boundary errors. For classification and detection one reports sensitivity, specificity, area under the receiver operating characteristic curve (AUROC), area under the precision-recall curve, and calibration (expected calibration error, Brier score). Reporting a single metric without uncertainty is insufficient.
\end{frame}

\begin{frame}{Transfer Learning and Self-supervised Pretraining}
    Annotated medical datasets are typically small. Transfer learning reuses weights trained on natural images (ImageNet) or on large unlabeled medical corpora. Self-supervised pretext tasks include masked image modelling, contrastive learning (SimCLR, MoCo), and restoration tasks specific to medical imaging such as context restoration and rotation prediction.

    Foundation models for medical imaging, for example MedSAM, RadDINO, and MI-Zero, provide general-purpose representations that are fine-tuned on small downstream datasets. This paradigm shifts compute budget from each task to a single pretraining run.
\end{frame}

\begin{frame}{nnU-Net: Self-configuring Segmentation Baseline}

\begin{frame}{Common Failure Modes}
    Deep networks fail silently on distributions they have not seen. Typical failure modes include:
    \begin{itemize}
        \item Shortcut learning, where the network exploits acquisition or site signatures rather than anatomy.
        \item Domain shift between sites, scanners, protocols, and patient populations.
        \item Class imbalance collapse, where the network predicts only the majority class.
        \item Confident errors on out-of-distribution inputs, including artefacts and rare pathologies.
    \end{itemize}
    Mitigations include careful data curation, domain generalisation techniques, uncertainty estimation, and out-of-distribution detection, all of which are active research topics in the medical imaging community.
\end{frame}

\begin{frame}{Uncertainty Estimation}
    Predictions in medical imaging should come with a measure of confidence. Common approaches are Monte Carlo dropout, deep ensembles (train several networks with different seeds), evidential deep learning, and conformal prediction.

    Aleatoric uncertainty reflects noise inherent to the data (e.g. partial volume effects at tissue boundaries). Epistemic uncertainty reflects limits of the model or training data and shrinks with more annotations. Only epistemic uncertainty is reducible, and distinguishing both components informs whether to acquire more data or to revise the model.
\end{frame}

\begin{frame}{Deployment, Regulation and Reproducibility}
    Translation from research code to clinical deployment involves monitoring for data drift, revalidation after software or hardware changes, integration into picture archiving and communication systems (PACS), and regulatory approval (CE mark, FDA clearance). Reproducibility requires version-controlled data splits, fixed random seeds, documented preprocessing, and release of trained weights and inference code.

    Standards such as MONAI Deploy, DICOM-based inference endpoints, and HL7 FHIR AI reporting are emerging to make these steps routine rather than bespoke.
\end{frame}

\begin{frame}{Practical Workflow}
    A deep learning project in medical imaging typically proceeds as follows:
    \begin{enumerate}
        \item Define the clinical question, the cohort, and the reference standard.
        \item Curate and preprocess data, harmonise resolution and orientation, anonymise.
        \item Split subjects into training, validation, and test sets at the patient level.
        \item Choose a baseline (nnU-Net for segmentation, a ResNet or ViT for classification).
        \item Train with strong augmentation and validated loss functions, monitor metrics.
        \item Evaluate on internal and external test sets, report uncertainty and failure modes.
        \item Iterate, and only after stable performance deploy with monitoring in place.
    \end{enumerate}
\end{frame}

\begin{frame}{Summary of Deep Learning for Medical Imaging}
    Convolutional and transformer-based networks turn image analysis into a representation-learning problem. Training combines a differentiable loss, backpropagation, and careful regularisation. The U-Net family remains the workhorse for segmentation, with nnU-Net as the standard baseline. Rigorous evaluation, domain-aware augmentation, uncertainty quantification, and reproducibility discipline are as important as architectural choices. The next lectures build on these tools for specific modalities and clinical tasks.
\end{frame}
